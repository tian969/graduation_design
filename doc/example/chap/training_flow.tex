\begin{figure}[htbp]
\centering
\begin{tikzpicture}[
  font=\sffamily\scriptsize,
  node distance=6mm and 10mm,
  box/.style={
    draw,
    rounded corners,
    align=center,
    inner sep=4pt,
    minimum width=26mm,
    minimum height=9mm,
    fill=blue!5
  },
  box2/.style={
    draw,
    rounded corners,
    align=center,
    inner sep=4pt,
    minimum width=26mm,
    minimum height=9mm,
    fill=green!5
  },
  box3/.style={
    draw,
    rounded corners,
    align=center,
    inner sep=4pt,
    minimum width=26mm,
    minimum height=9mm,
    fill=orange!5
  },
  data/.style={
    draw,
    rounded corners,
    align=center,
    inner sep=4pt,
    minimum width=28mm,
    minimum height=9mm,
    fill=gray!10
  },
  arr/.style={-Latex, line width=0.7pt},
  dasharr/.style={-Latex, line width=0.7pt, dashed}
]

% ===== 阶段1：数据准备 =====
\node[data] (input)
{多模态输入\\
$(x_q, x_i)$};

\node[data, right=of input] (label)
{类别标注\\
$x_c$};

\node[data, right=of label] (principle)
{安全原则映射\\
$p = \mathrm{P}(x_c)$};

\node[data, right=of principle] (rule)
{实例规则生成\\
$r^{\text{rule}} = \mathrm{R}(p, x_q)$};

% ===== 阶段2：监督微调 =====
\node[box, below=of principle] (sft_title)
{\textbf{阶段1：监督微调}\\
\textit{Reasoning-Aware SFT}};

\node[box, below=of sft_title] (cot_train)
{安全推理训练\\
$\mathrm{L}_{\text{cot}}$};

\node[box, right=of cot_train] (answer_train)
{回复生成训练\\
$\mathrm{L}_{\text{answer}}$};

\node[data, below=of cot_train] (sft_data)
{训练样本：\\
$(x, r^{\text{rule}}, z^*, y^*)$};

\node[data, below=of answer_train] (sft_loss)
{总损失：\\
$\mathrm{L}_{\text{SFT}} = \mathrm{L}_{\text{cot}} + \mathrm{L}_{\text{answer}}$};

\node[data, below=of sft_loss] (theta1)
{初始化参数：\\
$\theta_1$};

% ===== 阶段3：强化学习 =====
\node[box2, below=of theta1] (rl_title)
{\textbf{阶段2：在线强化学习}\\
\textit{Rule-Based RL}};

\node[box2, below=of rl_title] (policy)
{策略模型\\
$\pi_\theta(y \mid x)$};

\node[box2, right=of policy] (generate)
{生成回复\\
$y \sim \pi_\theta$};

\node[data, below=of generate] (reward)
{规则验证\\
$r(x,y) = r_{\text{safe}} \times r_{\text{help}}$};

\node[data, below=of reward] (gradient)
{策略梯度更新\\
$\nabla_\theta J(\theta)$};

\node[data, below=of gradient] (theta2)
{最终参数：\\
$\theta_2$};

% ===== 连接线：数据准备 =====
\draw[arr] (input) -- (label);
\draw[arr] (label) -- (principle);
\draw[arr] (principle) -- (rule);

% ===== 连接线：SFT阶段 =====
\draw[arr] (rule) -- (cot_train);
\draw[dasharr] (rule) -- (answer_train);
\draw[arr] (cot_train) -- (answer_train);
\draw[arr] (cot_train) -- (sft_data);
\draw[arr] (answer_train) -- (sft_loss);
\draw[arr] (sft_data) -- (sft_loss);
\draw[arr] (sft_loss) -- (theta1);

% ===== 连接线：RL阶段 =====
\draw[arr] (theta1) -- (policy);
\draw[arr] (policy) -- (generate);
\draw[arr] (generate) -- (reward);
\draw[arr] (reward) -- (gradient);
\draw[arr] (gradient) -- (theta2);
\draw[dasharr] (gradient) to[bend left=45] (policy);

% ===== 标注 =====
\node[right=0.5cm of sft_title, text=blue!70, font=\bfseries] {(1)};
\node[right=0.5cm of rl_title, text=green!60, font=\bfseries] {(2)};

\end{tikzpicture}
\caption{基于规则约束的两阶段安全训练流程。第一阶段通过监督微调学习安全推理能力，\\
建立规则理解—风险分析—安全生成的推理结构；第二阶段通过在线强化学习优化模型行为，\\
在规则约束下平衡安全性与有用性。}
\label{fig:training_flow}
\end{figure}
